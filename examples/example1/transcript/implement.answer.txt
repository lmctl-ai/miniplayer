import { UsageError } from './errors';

export type ParsedCommand =
  | { command: 'create'; name: string; text?: string }
  | { command: 'overwrite'; name: string; text?: string }
  | { command: 'show'; name: string }
  | { command: 'delete'; name: string }
  | { command: 'list' };

export function parseArgs(argv: string[]): ParsedCommand {
  const [command, name, text, ...rest] = argv;

  if (command === undefined) {
    throw new UsageError('Missing command');
  }

  switch (command) {
    case 'create':
      if (name === undefined) {
        throw new UsageError('Missing name for create');
      }
      if (rest.length > 0) {
        throw new UsageError('Too many arguments for create');
      }
      return { command: 'create', name, text };

    case 'overwrite':
      if (name === undefined) {
        throw new UsageError('Missing name for overwrite');
      }
      if (rest.length > 0) {
        throw new UsageError('Too many arguments for overwrite');
      }
      return { command: 'overwrite', name, text };

    case 'show':
      if (name === undefined) {
        throw new UsageError('Missing name for show');
      }
      if (text !== undefined || rest.length > 0) {
        throw new UsageError('Too many arguments for show');
      }
      return { command: 'show', name };

    case 'delete':
      if (name === undefined) {
        throw new UsageError('Missing name for delete');
      }
      if (text !== undefined || rest.length > 0) {
        throw new UsageError('Too many arguments for delete');
      }
      return { command: 'delete', name };

    case 'list':
      if (name !== undefined || text !== undefined || rest.length > 0) {
        throw new UsageError('Too many arguments for list');
      }
      return { command: 'list' };

    default:
      throw new UsageError(`Unknown command: ${command}`);
  }
}

export async function readStdin(): Promise<string> {
  return new Promise((resolve, reject) => {
    let data = '';

    process.stdin.setEncoding('utf8');
    process.stdin.on('data', (chunk: Buffer | string) => {
      data += chunk.toString();
    });
    process.stdin.on('end', () => {
      resolve(data);
    });
    process.stdin.on('error', (error) => {
      reject(error);
    });
  });
}

import * as fs from 'node:fs';
import type { Dirent } from 'node:fs';
import * as path from 'node:path';
import { DomainError } from './errors';

const VALID_NAME = /^[A-Za-z0-9_-]+$/;

export function validateName(name: string): void {
  if (!VALID_NAME.test(name)) {
    throw new DomainError(`Invalid note name: ${name}`);
  }
}

export function getNotePath(notesDir: string, name: string): string {
  validateName(name);
  return path.join(notesDir, `${name}.txt`);
}

export function ensureNotesDir(notesDir: string): void {
  try {
    fs.mkdirSync(notesDir, { recursive: true });
  } catch (error) {
    throw new DomainError(`Failed to create notes directory: ${errorMessage(error)}`);
  }
}

export function createNote(notesDir: string, name: string, text: string): void {
  const filePath = getNotePath(notesDir, name);
  ensureNotesDir(notesDir);

  try {
    fs.writeFileSync(filePath, text, { encoding: 'utf8', flag: 'wx' });
  } catch (error) {
    if (isNodeError(error) && error.code === 'EEXIST') {
      throw new DomainError(`Note already exists: ${name}`);
    }
    throw new DomainError(`Failed to create note: ${errorMessage(error)}`);
  }
}

export function overwriteNote(notesDir: string, name: string, text: string): void {
  const filePath = getNotePath(notesDir, name);
  ensureNotesDir(notesDir);

  if (!fs.existsSync(filePath)) {
    throw new DomainError(`Note not found: ${name}`);
  }

  try {
    fs.writeFileSync(filePath, text, { encoding: 'utf8' });
  } catch (error) {
    throw new DomainError(`Failed to overwrite note: ${errorMessage(error)}`);
  }
}

export function showNote(notesDir: string, name: string): string {
  const filePath = getNotePath(notesDir, name);

  try {
    return fs.readFileSync(filePath, 'utf8');
  } catch (error) {
    if (isNodeError(error) && error.code === 'ENOENT') {
      throw new DomainError(`Note not found: ${name}`);
    }
    throw new DomainError(`Failed to read note: ${errorMessage(error)}`);
  }
}

export function deleteNote(notesDir: string, name: string): void {
  const filePath = getNotePath(notesDir, name);

  try {
    fs.unlinkSync(filePath);
  } catch (error) {
    if (isNodeError(error) && error.code === 'ENOENT') {
      throw new DomainError(`Note not found: ${name}`);
    }
    throw new DomainError(`Failed to delete note: ${errorMessage(error)}`);
  }
}

export function listNotes(notesDir: string): string[] {
  let entries: Dirent[];

  try {
    entries = fs.readdirSync(notesDir, { withFileTypes: true });
  } catch (error) {
    if (isNodeError(error) && error.code === 'ENOENT') {
      return [];
    }
    throw new DomainError(`Failed to list notes: ${errorMessage(error)}`);
  }

  return entries
    .filter((entry) => entry.isFile() && entry.name.endsWith('.txt'))
    .map((entry) => entry.name.slice(0, -'.txt'.length))
    .sort();
}

function isNodeError(error: unknown): error is NodeJS.ErrnoException {
  return typeof error === 'object' && error !== null && 'code' in error;
}

function errorMessage(error: unknown): string {
  return error instanceof Error ? error.message : String(error);
}

import type { ParsedCommand } from './args';
import { readStdin } from './io';
import {
  createNote,
  deleteNote,
  listNotes,
  overwriteNote,
  showNote,
} from './notes';

export interface CommandDependencies {
  notesDir: string;
  stdout: { write(chunk: string): unknown };
  readStdin?: () => Promise<string>;
}

export async function runCommand(
  command: ParsedCommand,
  dependencies: CommandDependencies
): Promise<void> {
  const { notesDir, stdout } = dependencies;
  const read = dependencies.readStdin ?? readStdin;

  switch (command.command) {
    case 'create': {
      const text = command.text ?? (await read());
      createNote(notesDir, command.name, text);
      return;
    }

    case 'overwrite': {
      const text = command.text ?? (await read());
      overwriteNote(notesDir, command.name, text);
      return;
    }

    case 'show': {
      const content = showNote(notesDir, command.name);
      stdout.write(content);
      return;
    }

    case 'delete': {
      deleteNote(notesDir, command.name);
      return;
    }

    case 'list': {
      const names = listNotes(notesDir);
      if (names.length > 0) {
        stdout.write(`${names.join('\n')}\n`);
      }
      return;
    }
  }
}

import { parseArgs } from './args';
import { runCommand } from './commands';
import { getNotesDir } from './config';
import { AppError } from './errors';
import { readStdin } from './io';

export async function main(argv: string[] = process.argv.slice(2)): Promise<number> {
  try {
    const command = parseArgs(argv);
    const notesDir = getNotesDir();

    await runCommand(command, {
      notesDir,
      stdout: process.stdout,
      readStdin,
    });

    return 0;
  } catch (error) {
    if (error instanceof AppError) {
      process.stderr.write(`${error.message}\n`);
      return error.exitCode;
    }

    const message =
      error instanceof Error ? error.stack ?? error.message : String(error);
    process.stderr.write(`${message}\n`);
    return 1;
  }
}

if (require.main === module) {
  void main().then((exitCode) => {
    process.exitCode = exitCode;
  });
}

import test from 'node:test';
import assert from 'node:assert/strict';
import { spawnSync } from 'node:child_process';
import * as fs from 'node:fs';
import * as os from 'node:os';
import * as path from 'node:path';

const cliPath = path.resolve(process.cwd(), 'src/cli.ts');
const tsxCli = path.resolve(
  process.cwd(),
  'node_modules',
  'tsx',
  'dist',
  'cli.mjs'
);

function runCli(
  args: string[],
  options: { input?: string; env?: NodeJS.ProcessEnv } = {}
) {
  return spawnSync(process.execPath, [tsxCli, cliPath, ...args], {
    input: options.input,
    encoding: 'utf8',
    env: { ...process.env, ...options.env },
  });
}

function tempDir(): string {
  return fs.mkdtempSync(path.join(os.tmpdir(), 'notes-cli-cli-'));
}

test('cli create/list/show/overwrite/delete', () => {
  const dir = tempDir();
  const env = { NOTES_DIR: dir };

  try {
    let result = runCli(['create', 'alpha', 'hello'], { env });
    assert.equal(result.status, 0);
    assert.equal(result.stdout, '');
    assert.equal(result.stderr, '');

    result = runCli(['list'], { env });
    assert.equal(result.status, 0);
    assert.equal(result.stdout, 'alpha\n');

    result = runCli(['show', 'alpha'], { env });
    assert.equal(result.status, 0);
    assert.equal(result.stdout, 'hello');

    result = runCli(['overwrite', 'alpha', 'world'], { env });
    assert.equal(result.status, 0);

    result = runCli(['show', 'alpha'], { env });
    assert.equal(result.status, 0);
    assert.equal(result.stdout, 'world');

    result = runCli(['delete', 'alpha'], { env });
    assert.equal(result.status, 0);

    result = runCli(['list'], { env });
